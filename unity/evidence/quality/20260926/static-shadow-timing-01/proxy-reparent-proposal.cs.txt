// Proposed narrow EDITOR operation for root, after all static timing is complete.
// This text is outside Assets and has NOT been compiled or executed.
// Read actual objects first and preserve the before/after JSON evidence.
var player = UnityEngine.Object.FindAnyObjectByType<AthenHill.PlayerMotor>();
if (!player || !player.visual || player.visual.parent != player.transform)
    throw new System.InvalidOperationException("Unexpected player visual topology.");
var proxy = player.transform.Find("Player shadow proxy");
if (!proxy || proxy.GetComponentsInChildren<UnityEngine.Collider>(true).Length != 0)
    throw new System.InvalidOperationException("Proxy missing, already moved, or contains collision.");
var renderer = proxy.GetComponent<UnityEngine.MeshRenderer>();
if (!renderer || !renderer.enabled ||
    renderer.shadowCastingMode != UnityEngine.Rendering.ShadowCastingMode.ShadowsOnly)
    throw new System.InvalidOperationException("Unexpected proxy renderer state.");
if ((player.visual.localScale - UnityEngine.Vector3.one).sqrMagnitude > 1e-8f ||
    player.visual.localPosition.sqrMagnitude > 1e-8f ||
    UnityEngine.Quaternion.Angle(player.visual.localRotation, UnityEngine.Quaternion.identity) > .001f)
    throw new System.InvalidOperationException("Inspect visual placement before reparenting.");

var position = proxy.position;
var rotation = proxy.rotation;
var scale = proxy.lossyScale;
var material = renderer.sharedMaterial;
var mesh = proxy.GetComponent<UnityEngine.MeshFilter>().sharedMesh;
var receiveShadows = renderer.receiveShadows;
var active = proxy.gameObject.activeSelf;
var objectId = proxy.gameObject.GetInstanceID();

// Undo.SetTransformParent preserves the world transform and records the hierarchy change.
UnityEditor.Undo.SetTransformParent(proxy, player.visual, "Interpolate player shadow proxy");

if ((proxy.position - position).sqrMagnitude > 1e-8f ||
    UnityEngine.Quaternion.Angle(proxy.rotation, rotation) > .001f ||
    (proxy.lossyScale - scale).sqrMagnitude > 1e-8f ||
    proxy.gameObject.GetInstanceID() != objectId ||
    renderer.sharedMaterial != material ||
    proxy.GetComponent<UnityEngine.MeshFilter>().sharedMesh != mesh ||
    renderer.receiveShadows != receiveShadows || proxy.gameObject.activeSelf != active ||
    renderer.shadowCastingMode != UnityEngine.Rendering.ShadowCastingMode.ShadowsOnly)
    throw new System.InvalidOperationException("Reparent changed more than hierarchy; undo and inspect.");

// Save only after evidence checks. Keep this as a scene added-object override;
// do not apply it to the source MeshyPlayer prefab or regenerate the player.
UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(player.gameObject.scene);
UnityEditor.SceneManagement.EditorSceneManager.SaveScene(player.gameObject.scene);
