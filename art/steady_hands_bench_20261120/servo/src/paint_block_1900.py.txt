# The paint block of servo/build.py as it was after the 19:00 surface fix round (baked 18:48, published 18:52), kept so
# the previous look can be put back: paste it over the block between "# Warden orange and the bare alloy" and the
# sv_bell material. The baked export of that state is in servo/export_prev_1848/. sv.py lost nothing (add_wear,
# add_scratches and add_chips are still there).
# Warden orange and the bare alloy under it are the programme's shared colours (SEAM.md section 8, linear albedo)
ORANGE = sv.lin((0.60, 0.20, 0.076))
BARE = (sv.lin((0.55, 0.53, 0.50)), 0.5, 1.0)
PRIMER = ((0.26, 0.13, 0.07), 0.85)
PAINT = dict(metal=0.0, rough=0.58, var=0.09, edge=(BARE[0], 1.0, 0.5, 1.0), dirt=DIRT, rust=((0.33, 0.15, 0.06), 0.08),
             bumps=[('noise', 700, 0.3), ('stipple', 2600, 0.14)], edge_width=0.0021, tone2=(sv.lin((0.40, 0.12, 0.045)), 42, 0.5, 0.74, 0.34), grime=0.4)     # tone2: handling and oil stains
LID_O = P(8, 0, SEAM)


def fld(c, r, o=(0, 0, 0)):
    """A wear field: centre and radii in millimetres (world), moved into the object space of a part whose origin is o."""
    return (P(*c) - Vector(o), (r[0] * MM, r[1] * MM, r[2] * MM))


# where the paint is worn through: few, large, where the thing was handled, clamped, stood and dragged
WEAR = {
    'sv_paint': [fld((6, -23.5, 15), (31, 7, 10)),                                           # the side it lay on, dragged
                 fld((-15, 27, 1.5), (9, 7, 4)), fld((27, -27, 1.5), (10, 8, 4)), fld((-17, 0, 2), (4, 14, 4)),      # the foot's corners and front edge
                 fld((6, 23, 3), (8, 5, 3)), fld((6, -23, 3), (7, 4.5, 3)),                   # round the bolt slots
                 fld((-2, 31, 28.5), (6, 4, 4)), fld((19, 31, 28.5), (6, 4, 4)), fld((8, -31, 28.5), (6, 4, 4)),     # the clip seats
                 fld((-16.5, 33, 27), (5, 4, 6)), fld((32.5, -33, 27), (5, 4, 6)),            # two screw bosses
                 fld((34, 12, 8), (6, 9, 6))],
    'sv_lidpaint': [fld(c, r, LID_O) for c, r in (
        ((-12, 5, 55.0), (2.4, 8, 2.6)), ((28, -3, 55.2), (2.4, 7, 2.6)),                    # two rubbed rib crests
        ((-2, 30.5, 32.5), (7, 3.5, 3)), ((19, 30.5, 32.5), (7, 3.5, 3)), ((8, -30.5, 32.5), (7, 3.5, 3)),           # the flange under the clips
        ((22, -30.8, 32.5), (10, 2.5, 2.5)),                                                  # a length of flange edge
        ((-16.5, 30.5, 33.4), (5.5, 5.5, 2.5)), ((32.5, 30.5, 33.4), (5, 5, 2.5)), ((32.5, -30.5, 33.4), (5, 5, 2.5)), ((-16.5, -30.5, 33.4), (5.5, 5.5, 2.5)),   # round the screws
        ((-4, -17, 49), (9, 7, 6)), ((3, -20.5, 45), (7, 5, 5)))],                           # one field on the shoulder, two lobes
}
BURNT = ((0.20, 0.10, 0.06), 0.8)             # paint darkened by heat
SOOT = ((0.035, 0.03, 0.027), 0.95)
for nm in ('sv_paint', 'sv_lidpaint'):
    M[nm] = MT.make(nm, ORANGE, **PAINT)
    if nm == 'sv_lidpaint':      # the stencilled serial, on the lid's -Y shoulder between two ribs
        ta = math.radians(113)
        sv.add_stencil(M[nm], os.path.join(HERE, 'src', 'stencil_8372.png'), P(18, RO * math.cos(ta), ZA + RO * math.sin(ta)) - LID_O,
                       (1, 0, 0), (0, math.sin(ta), -math.cos(ta)), 0.0150, 0.0040, (0.80, 0.76, 0.66))
    sv.add_wear(M[nm], WEAR[nm], BARE[0], BARE[1], BARE[2], rim=PRIMER, noise_scale=58.0, soft=1.1)
    sv.add_scratches(M[nm], BARE[0], BARE[1], BARE[2], scale=(7.0, 950.0, 950.0), lo=0.70, hi=0.72)
    sv.add_chips(M[nm], BARE[0], BARE[1], BARE[2], scale=100.0, lo=0.70, hi=0.72, fine=0.0, rim=PRIMER, depth=-0.7)      # a rare stray flake
    sv.add_streaks(M[nm], (0.20, 0.14, 0.09), 0.85, axis='Z', scale=(300.0, 300.0, 26.0), lo=0.56, hi=0.8, amount=0.45)
    # the scorch at the rear +Y corner: paint darkened and blistered in a wide soft field, soot thickening to black
    oo = LID_O if nm == 'sv_lidpaint' else Vector((0, 0, 0))
    sv.add_patch(M[nm], P(32, 16, 47) - oo, 0.034, BURNT[0], BURNT[1], 0.0, noise_scale=150, soft=0.45, bump=2.6, inner=0.12, amount=0.9)
    sv.add_patch(M[nm], P(34, 17, 47) - oo, 0.024, SOOT[0], SOOT[1], 0.0, noise_scale=210, soft=0.5, inner=0.05)
    sv.add_dust(M[nm], DUST, veil=0.035, cav=1.9, up=1.5)
